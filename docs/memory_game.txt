CREATE DATABASE IF NOT EXISTS memory_game;
USE memory_game;

CREATE TABLE JUGADORES (
    ID INT AUTO_INCREMENT PRIMARY KEY,
    nombre VARCHAR(100) NOT NULL,
    apellido VARCHAR(100) NOT NULL
);

CREATE TABLE CONJUNTOS_DE_CARTAS (
    ID INT AUTO_INCREMENT PRIMARY KEY,
    cantidad_pares INT NOT NULL,
    nombre VARCHAR(100) NOT NULL,
    tema VARCHAR(100),
    sprite_base VARCHAR(255)
);

CREATE TABLE PARTIDA (
    ID INT AUTO_INCREMENT PRIMARY KEY,
    nivel_dificultad VARCHAR(50) NOT NULL,
    fecha_inicio DATETIME NOT NULL,
    fecha_fin DATETIME,
    tiempo_total TIME,

    JUGADORES_ID INT NOT NULL,

    CONSTRAINT FK_PARTIDA_JUGADOR
        FOREIGN KEY (JUGADORES_ID)
        REFERENCES JUGADORES(ID)
        ON UPDATE CASCADE
        ON DELETE CASCADE
);

CREATE TABLE TABLEROS_GENERADOS (
    ID INT AUTO_INCREMENT PRIMARY KEY,
    CDC_ID INT NOT NULL,
    PARTIDA_ID INT NOT NULL,

    estado_inicial VARCHAR(50),
    tamaño_columnas INT NOT NULL,
    tamaño_filas INT NOT NULL,

    CONSTRAINT FK_TABLERO_CONJUNTO
        FOREIGN KEY (CDC_ID)
        REFERENCES CONJUNTOS_DE_CARTAS(ID)
        ON UPDATE CASCADE
        ON DELETE RESTRICT,

    CONSTRAINT FK_TABLERO_PARTIDA
        FOREIGN KEY (PARTIDA_ID)
        REFERENCES PARTIDA(ID)
        ON UPDATE CASCADE
        ON DELETE CASCADE,

    -- La relación CONTENER es 1:1
    CONSTRAINT UQ_TABLERO_PARTIDA
        UNIQUE (PARTIDA_ID)
);

CREATE TABLE PARES (
    ID INT AUTO_INCREMENT PRIMARY KEY,
    CDC_ID INT NOT NULL,
    par_id INT NOT NULL,
    valor_pareja VARCHAR(100),
    imagen VARCHAR(255),

    CONSTRAINT FK_PAR_CONJUNTO
        FOREIGN KEY (CDC_ID)
        REFERENCES CONJUNTOS_DE_CARTAS(ID)
        ON UPDATE CASCADE
        ON DELETE CASCADE
);

CREATE TABLE RESULTADO_FINAL (
    ID INT AUTO_INCREMENT PRIMARY KEY,

    PARTIDA_ID INT NOT NULL,

    intentos_totales INT DEFAULT 0,
    puntaje_eficiencia DECIMAL(10,2) DEFAULT 0,
    tiempo_tardado TIME,
    fallos INT DEFAULT 0,
    aciertos INT DEFAULT 0,

    CONSTRAINT FK_RESULTADO_PARTIDA
        FOREIGN KEY (PARTIDA_ID)
        REFERENCES PARTIDA(ID)
        ON UPDATE CASCADE
        ON DELETE CASCADE,

    -- PARTIDA_RESULTADO_FINAL es 1:1
    CONSTRAINT UQ_RESULTADO_PARTIDA
        UNIQUE (PARTIDA_ID)
);

CREATE TABLE RANKING (
    ID INT AUTO_INCREMENT PRIMARY KEY,

    PARTIDAS_ID INT NOT NULL,

    puntaje DECIMAL(10,2) NOT NULL,
    dificultad_del_nivel VARCHAR(50),
    mejor_tiempo TIME,

    CONSTRAINT FK_RANKING_PARTIDA
        FOREIGN KEY (PARTIDAS_ID)
        REFERENCES PARTIDA(ID)
        ON UPDATE CASCADE
        ON DELETE CASCADE
);

CREATE TABLE REGISTROS_INTENTO_DE_VOLTEO (
    ID INT AUTO_INCREMENT PRIMARY KEY,

    PARTIDA_ID INT NOT NULL,

    carta_id INT,
    carta1_id INT,
    `timestamp` DATETIME NOT NULL,
    fue_acierto BOOLEAN NOT NULL DEFAULT FALSE,

    CONSTRAINT FK_REGISTRO_PARTIDA
        FOREIGN KEY (PARTIDA_ID)
        REFERENCES PARTIDA(ID)
        ON UPDATE CASCADE
        ON DELETE CASCADE
);
